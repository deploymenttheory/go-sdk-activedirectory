function Write-FreshControlISORecord([string]$Path,$Value){
 $target=Assert-LinuxPath $Path
 $data=[Text.UTF8Encoding]::new($false).GetBytes(($Value|ConvertTo-Json -Depth 14))
 $file=[IO.File]::Open($target,[IO.FileMode]::CreateNew,[IO.FileAccess]::Write,[IO.FileShare]::None)
 try{$file.Write($data,0,$data.Length)}finally{$file.Dispose()}
 return $target
}
function Get-ControlISODeviceState {
 [pscustomobject][ordered]@{
  dvd=@(Get-VMDvdDrive -VM $vm|Sort-Object Id|ForEach-Object{[pscustomobject][ordered]@{id=$_.Id.ToString();path=$_.Path;controller=$_.ControllerNumber;location=$_.ControllerLocation;type=$_.ControllerType.ToString()}})
  hdd=@(Get-VMHardDiskDrive -VM $vm|Sort-Object Id|ForEach-Object{[pscustomobject][ordered]@{id=$_.Id.ToString();path=$_.Path;controller=$_.ControllerNumber;location=$_.ControllerLocation;type=$_.ControllerType.ToString()}})
  scsi=@(Get-VMScsiController -VM $vm|Sort-Object ControllerNumber|ForEach-Object{[pscustomobject]@{id=$_.Id.ToString();number=$_.ControllerNumber}})
  nics=@(Get-VMNetworkAdapter -VM $vm|Sort-Object Id|ForEach-Object{[pscustomobject]@{id=$_.Id.ToString();switch_id=$_.SwitchId.ToString();mac=$_.MacAddress}})
 }
}
function Assert-ControlISOMembers($Entries){
 if(@($Entries).Count-ne2){throw 'Public ISO must contain exactly two members.'}
 foreach($entry in @($Entries)){
  if($entry.PSIsContainer-or$entry.Name-cnotin@('peer.exe','manifest.json')-or$entry.Attributes-band[IO.FileAttributes]::ReparsePoint){throw 'Unexpected/private/autorun ISO member.'}
 }
 if(@($Entries|Select-Object -ExpandProperty Name -Unique).Count-ne2){throw 'Duplicate public ISO member.'}
}
function Convert-ControlISOToWSLPath([string]$Path){
 $absolute=Assert-LinuxPath $Path
 if($absolute-cnotmatch'\A([A-Za-z]):\\'){throw 'ISO builder requires an owned local drive path.'}
 return '/mnt/'+$Matches[1].ToLowerInvariant()+'/'+$absolute.Substring(3).Replace('\','/')
}
function Get-ControlISOVolumeLabel([string]$MediaId){
 if($MediaId-cnotmatch'\A[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}\z'){throw 'Invalid public media identity.'}
 # Joliet stores this 32-byte field as UCS-2, leaving only 16 characters.
 return 'AD'+$MediaId.Replace('-','').Substring(0,14).ToUpperInvariant()
}
function Assert-ControlISOLabels([string]$Path,[string]$Expected){
 if($Expected-cnotmatch'\A[A-Z0-9]{1,16}\z'){throw 'Expected ISO label is not Joliet-safe.'}
 $file=[IO.File]::OpenRead((Assert-LinuxPath $Path));$primary=@();$joliet=@();$terminated=$false
 try{
  for($sector=16;$sector-lt64;$sector++){
   $buffer=[byte[]]::new(2048);$file.Position=$sector*2048
   if($file.Read($buffer,0,$buffer.Length)-ne2048-or[Text.Encoding]::ASCII.GetString($buffer,1,5)-cne'CD001'-or$buffer[6]-ne1){throw 'Invalid ISO volume descriptor.'}
   if($buffer[0]-eq255){$terminated=$true;break}
   if($buffer[0]-eq1){$primary+= [Text.Encoding]::ASCII.GetString($buffer,40,32).TrimEnd([char]32,[char]0)}
   if($buffer[0]-eq2){
    $escape=([BitConverter]::ToString($buffer,88,3)).Replace('-','')
    if($escape-cnotin@('252F40','252F43','252F45')){throw 'Unexpected secondary ISO encoding.'}
    $joliet+= [Text.Encoding]::BigEndianUnicode.GetString($buffer,40,32).TrimEnd([char]32,[char]0)
   }
  }
 }finally{$file.Dispose()}
 if(-not$terminated-or$primary.Count-ne1-or$joliet.Count-ne1-or$primary[0]-cne$Expected-or$joliet[0]-cne$Expected){throw 'Actual primary/Joliet ISO labels differ from the exact expected label.'}
 return [pscustomobject]@{primary_label=$primary[0];joliet_label=$joliet[0];verified=$true}
}

function Save-ControlISOJournal {$journal|ConvertTo-Json -Depth 14|Set-Content -LiteralPath $journalPath -Encoding utf8}
