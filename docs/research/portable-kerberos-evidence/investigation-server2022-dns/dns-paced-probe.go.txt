package main

import (
	"context"
	"encoding/json"
	"net"
	"os"
	"time"
)

func main() {
	enc := json.NewEncoder(os.Stdout)
	for n := range 30 {
		ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
		start := time.Now()
		addresses, err := net.DefaultResolver.LookupIPAddr(ctx, "adsdk22dc1.ad.sdk.test")
		row := map[string]any{"attempt": n, "milliseconds": time.Since(start).Milliseconds(), "addresses": addresses}
		if err != nil {
			row["error"] = err.Error()
		}
		cancel()
		_ = enc.Encode(row)
        time.Sleep(time.Second)
	}
}
