   [pscustomobject]@{started=$started;result=$result;active_installers=$active;package_state=if($package){$package.PackageState.ToString()}else{'not-queried-while-installer-active'};boot=(Get-CimInstance Win32_OperatingSystem).LastBootUpTime.ToUniversalTime().ToString('o')}
  }
  if($Action-eq'Reconcile'){$record=Write-PublicRecord ('reconcile-'+$KB+'-'+$target.Name) $state;$state|Add-Member -NotePropertyName record -NotePropertyValue $record;$state;return}
  if($state.active_installers.Count-ne0-or-not$state.result-or$state.result.exit_code-notin@(0,3010)-or$state.package_state-notin@('Installed','InstallPending')){throw 'Cannot reboot while the installer is active or unresolved.'}
  $peerRole=if($Role-ceq'dc1'){'dc2'}else{'dc1'}
  $peer=Get-DCHealth $peerRole
  if(-not$peer.full_five_nc_healthy-or$peer.pending_reboot.cbs-or$peer.pending_reboot.windows_update){throw 'Partner is not ready to carry the domain during this reboot.'}
  Invoke-Command -Session $session -ArgumentList $guestDirectory,$profile.id,$target.Id.ToString(),$state.boot,$KB -ScriptBlock{
   param($directory,$profileID,$vmID,$boot,$kb)
   $path=Join-Path $directory 'controlled-reboot-intent.json'
   if(Test-Path -LiteralPath $path){throw 'A controlled reboot intent already exists: reconcile post-boot state; do not issue another restart.'}
   @{profile_id=$profileID;vm_id=$vmID;kb=$kb;boot_before=$boot;requested_utc=[DateTime]::UtcNow.ToString('o');automatic_retry=$false}|ConvertTo-Json|Set-Content -LiteralPath $path -Encoding utf8
  }
  $rebootRecord=[pscustomobject]@{profile_id=$profile.id;vm_id=$target.Id.ToString();computer=$target.Name;kb=$KB;package=$state.started.package;boot_before=$state.boot;requested_utc=[DateTime]::UtcNow.ToString('o');method='guest shutdown /r /t 0';installer_exited=$true;peer_full_five_nc_healthy=$true;force_power_cycle_used=$false}
  $null=Write-PublicRecord ('reboot-started-'+$KB+'-'+$target.Name) $rebootRecord
  try{
   Invoke-Command -Session $session -ScriptBlock{& "$env:WINDIR\System32\shutdown.exe" '/r' '/t' '0' '/d' 'p:2:17' '/c' 'Controlled SDK2019 cumulative update reboot after installer exit';if($LASTEXITCODE-ne0){throw 'Controlled guest reboot command failed.'}}
  }catch{Write-Host ('Reboot signal/session transition: '+$_.Exception.GetType().Name)}
  $deadline=[DateTime]::UtcNow.AddMinutes(45);$ready=$false;$lastFailure=''
  while([DateTime]::UtcNow-lt$deadline){
   $probe=$null
   try{
    $vm=Get-Owned2019VM $Role
    if($vm.State-ne'Running'){throw 'VM is not currently running during controlled reboot.'}
    $probe=New-PSSession -VMId $target.Id -Credential $admin -ErrorAction Stop
    $boot=Invoke-Command -Session $probe -ScriptBlock{
     Import-Module ActiveDirectory
     [pscustomobject]@{boot=(Get-CimInstance Win32_OperatingSystem).LastBootUpTime.ToUniversalTime().ToString('o');services=@(Get-Service NTDS,DNS,ADWS,Netlogon,Kdc,DFSR|Where-Object Status -NE Running).Count;synchronized=[string](Get-ADRootDSE -Server localhost).isSynchronized;shares=@(Get-SmbShare|Where-Object Name -in @('SYSVOL','NETLOGON')).Count}
    }
    if($boot.boot-cne$state.boot-and$boot.services-eq0-and$boot.synchronized-ieq'TRUE'-and$boot.shares-eq2){$ready=$true;break}
   }catch{$lastFailure=$_.Exception.GetType().Name}
   finally{if($probe){Remove-PSSession $probe}}
   Write-Host ([DateTime]::UtcNow.ToString('o')+' '+$target.Name+' controlled reboot/readiness pending; '+$lastFailure)
   Start-Sleep -Seconds 10
  }
  if(-not$ready){$null=Write-PublicRecord ('reboot-unresolved-'+$KB+'-'+$target.Name) $rebootRecord;throw 'Controlled reboot is still unresolved; no power cycle or servicing retry performed.'}
  $after=Get-DCHealth $Role $true
  $peerAfter=Get-DCHealth $peerRole $true
  $policySame=($state.started.policies_before|ConvertTo-Json -Depth 6 -Compress)-ceq($after.policies|ConvertTo-Json -Depth 6 -Compress)
  $package=@($after.packages|Where-Object name -CEQ $state.started.package)
  $verified=$after.build-ceq$inspection.expected_build-and$after.boot-cne$state.boot-and$after.full_five_nc_healthy-and$peerAfter.full_five_nc_healthy-and
   -not$after.pending_reboot.cbs-and-not$after.pending_reboot.windows_update-and$after.pending_reboot.pending_file_renames-eq0-and$policySame-and$package.Count-eq1-and$package[0].state-ceq'Installed'
  $final=[pscustomobject]@{verified=$verified;computer=$target.Name;kb=$KB;expected_build=$inspection.expected_build;policy_values_unchanged=$policySame;after=$after;peer_after=$peerAfter;restore_tested=$false;cbt_enforcement_qualified=$false}
  $record=Write-PublicRecord ('post-reboot-'+$KB+'-'+$target.Name) $final
  [pscustomobject]@{record=$record;verified=$verified;computer=$target.Name;build=$after.build;all_five_nc_healthy=$after.full_five_nc_healthy;peer_all_five_nc_healthy=$peerAfter.full_five_nc_healthy;policy_values_unchanged=$policySame;pending=$after.pending_reboot;cbt_enforcement_qualified=$false}|ConvertTo-Json -Depth 7
  if(-not$verified){throw 'Independent post-reboot package/build/five-NC/policy validation failed; stop further servicing writes.'}
  return
 }
}finally{Remove-PSSession $session}
