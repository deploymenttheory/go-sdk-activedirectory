import json
import os
import pathlib

owner = pathlib.Path('/etc/adsdk-linux-owner').read_text().strip()
if owner != '6df04253-32aa-4dd4-8031-1dd9174ca405' or os.geteuid() != 0:
    raise RuntimeError('Owned Linux administrator context differs')
path = '/etc/resolv.conf'
original = '../run/systemd/resolve/stub-resolv.conf'
if not os.path.islink(path) or os.readlink(path) != '/run/systemd/resolve/resolv.conf':
    raise RuntimeError('Temporary direct resolver differs; reconcile before mutation')
stub = pathlib.Path('/run/systemd/resolve/stub-resolv.conf')
if not stub.is_file() or stub.is_symlink():
    raise RuntimeError('Original maintained stub resolver is unavailable')
temporary = '/etc/.adsdk-resolv-2022-restore.tmp'
if os.path.lexists(temporary):
    raise RuntimeError('Restoration temporary entry exists; no replay')
os.symlink(original, temporary)
os.replace(temporary, path)
if os.readlink(path) != original or os.path.realpath(path) != str(stub):
    raise RuntimeError('Independent original resolver link readback differs')
print(json.dumps(dict(linux_owner=owner, original_link=original,
                     restored_link=os.readlink(path), restored=True,
                     directory_calls=0, clock_changed=False)))
