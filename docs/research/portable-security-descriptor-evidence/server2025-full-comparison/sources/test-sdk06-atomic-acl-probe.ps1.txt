$script:Checks=0
function Assert-Probe($Condition,[string]$Label) {if (-not $Condition) {throw ('FAIL: '+$Label)};$script:Checks++}
function Assert-ProbeRejected([scriptblock]$Call,[string]$Label) {$rejected=$false;try {$null=& $Call} catch {$rejected=$true};Assert-Probe $rejected $Label}
function New-MockStatus([Nullable[int]]$Code) {return [pscustomobject]@{response_received=($null -ne $Code);code=$Code;diagnostic=$null;request_submitted=$true}}
function New-MockProbe([int]$Mask=7) {
    $source=Get-Sdk06AtomicSource;$id='sdk06-cas2-'+[guid]::NewGuid().ToString('N').Substring(0,18)
    $state=New-Sdk06AtomicState $id $Mask 'adsdk25dc1.ad.sdk.test' 'adsdk25dc2.ad.sdk.test' $source
    $journal=Initialize-Sdk06AtomicJournal $state $OutputBase $source
    $sd=[Security.AccessControl.RawSecurityDescriptor]::new('O:SYG:SYD:(A;;GA;;;SY)(A;;GA;;;BA)')
    $bytes=[byte[]]::new($sd.BinaryLength);$sd.GetBinaryForm($bytes,0)
    $world=[pscustomobject]@{entries=@{};descriptors=@{};adds=0;cas=0;setup=0;deletes=0;calls=0;strict_known=$true;permissive=$false;expired=$false;compare_mode='normal';read_code=0;stale_wrong=$false;uncertain_cas=$false;delete_status=0;peer_code=32;source=$source;journal=$journal;state=$state;baseline=$bytes}
    $before={
        $world.calls++
        if ($world.expired) {throw 'Mock action budget elapsed.'}
    }.GetNewClosure()
    $intent={param($phase)
        $events=@(Get-ChildItem -LiteralPath $world.journal.directory -Filter 'event-*.json'|Sort-Object Name)
        $latest=Get-Content -LiteralPath $events[-1].FullName -Raw|ConvertFrom-Json
        if ($latest.phase -cne $phase -or -not (Test-Path -LiteralPath (Join-Path $world.journal.directory 'probe-source.ps1.txt'))) {throw 'Mutation reached mock without durable source and intent.'}
    }.GetNewClosure()
    $ops=[pscustomobject]@{
        CheckBudget=$before
        Context={param($selected) & $before;[pscustomobject]@{server=$selected;domain_dn='DC=ad,DC=sdk,DC=test';adds=$true;sd_flags_supported=$true;permissive_known=$world.strict_known;permissive_modify=$world.permissive}}.GetNewClosure()
        Get={param($selected,$guid,$dn)
            & $before
            $found=@($world.entries.Values|Where-Object {if ($guid) {$_.guid -eq $guid} else {$_.dn -ceq $dn}})
            $code=32;$object=$null
            if ($found.Count -eq 1) {$code=$world.read_code;$object=$found[0]}
            if ($selected -ceq 'adsdk25dc2.ad.sdk.test' -and -not $found.Count) {$code=$world.peer_code}
            [pscustomobject]@{status=(New-MockStatus $code);object=$object;objects=$found}
        }.GetNewClosure()
        List={param($selected,$dn) & $before;$items=@($world.entries.Values|Where-Object {$_.dn -ceq $dn -or $_.dn.EndsWith(','+$dn,[StringComparison]::Ordinal)});$code=0;if (-not $items.Count) {$code=32};[pscustomobject]@{status=(New-MockStatus $code);objects=$items}}.GetNewClosure()
        Children={param($selected,$dn) & $before;[pscustomobject]@{status=(New-MockStatus 0);objects=@($world.entries.Values|Where-Object {$_.dn.EndsWith(','+$dn,[StringComparison]::Ordinal)})}}.GetNewClosure()
        Markers={param($selected,$current) & $before;[pscustomobject]@{status=(New-MockStatus 0);objects=@($world.entries.Values)}}.GetNewClosure()
        Add={param($selected,$expected)
            & $before;& $intent 'before-owned-ou-add';$world.adds++
            $guid=[guid]::NewGuid().ToString('D');$world.entries[$guid]=[pscustomobject]@{dn=$expected.dn;guid=$guid;kind='organizationalUnit';marker=$expected.marker}
            $world.descriptors[$guid]=[Convert]::ToBase64String($world.baseline)
            New-MockStatus 0
        }.GetNewClosure()
        ReadSD={param($selected,$expected,$mask)
            & $before;$raw=[Convert]::FromBase64String($world.descriptors[$expected.guid])
            if ($mask -eq 4) {$part=[Security.AccessControl.RawSecurityDescriptor]::new($raw,0);$part.Owner=$null;$part.Group=$null;$raw=[byte[]]::new($part.BinaryLength);$part.GetBinaryForm($raw,0)}
            [pscustomobject]@{status=(New-MockStatus 0);object=$world.entries[$expected.guid];bytes=$raw}
        }.GetNewClosure()
        Compare={param($selected,$expected,$raw)
            & $before
            if ($world.compare_mode -ceq 'unknown') {return New-MockStatus $null}
            if ($world.compare_mode -ceq 'unsupported') {return New-MockStatus 18}
            if ($world.stale_wrong -and $world.setup -gt 0) {return New-MockStatus 6}
            $code=5;if ([Convert]::ToBase64String($raw) -ceq $world.descriptors[$expected.guid]) {$code=6};New-MockStatus $code
        }.GetNewClosure()
        Modify={param($selected,$expected,$shape,$old,$new)
            & $before
            if ($shape -ceq 'fixture-replace') {& $intent 'before-independent-fixture-replace';$world.setup++;$world.descriptors[$expected.guid]=[Convert]::ToBase64String($new);return New-MockStatus 0}
            & $intent 'before-strict-cas-submit';$world.cas++
            if ($world.uncertain_cas) {return New-MockStatus $null}
            if ([Convert]::ToBase64String($old) -cne $world.descriptors[$expected.guid]) {return New-MockStatus 16}
            $world.descriptors[$expected.guid]=[Convert]::ToBase64String($new);New-MockStatus 0
        }.GetNewClosure()
        Delete={param($selected,$expected)
            & $before;& $intent 'before-owned-leaf-delete';$world.deletes++
            if ($world.delete_status -eq 0) {$world.entries.Remove($expected.guid);$world.descriptors.Remove($expected.guid)}
            New-MockStatus $world.delete_status
        }.GetNewClosure()
    }
    return [pscustomobject]@{world=$world;state=$state;journal=$journal;ops=$ops}
}

Assert-ProbeRejected {Invoke-Sdk06AtomicPrepareCore $m.state $m.journal $m.ops} 'Existing directory namespace cannot be adopted';Assert-Probe ($m.world.adds -eq 0 -and $m.world.cas -eq 0 -and $m.world.deletes -eq 0) 'Existing directory namespace receives no writes'
