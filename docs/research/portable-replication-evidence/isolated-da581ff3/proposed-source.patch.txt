+// validateFragment checks wire-owned lengths before slicing or authentication.
+func (h Header) validateFragment(available int) error {
+	length := int(h.FragLength)
+	if h.RPCVersion != 5 || h.RPCVersionMinor != 0 || h.PacketDRep != ndr.DefaultDataRepresentation {
+		return fmt.Errorf("%w: unsupported RPC version or representation", ErrMalformedFragment)
+	}
+	if length < HeaderSize || length > available {
+		return fmt.Errorf("%w: invalid complete-fragment length", ErrMalformedFragment)
+	}
+	if h.AuthLength != 0 && int(h.AuthLength)+SecurityTrailerSize > length-HeaderSize {
+		return fmt.Errorf("%w: authentication exceeds the fragment", ErrMalformedFragment)
+	}
+	return nil
+}

+// validatePDU checks fixed headers and authentication layout before use.
+func (h Header) validatePDU(available int) error {
+	if err := h.validateFragment(available); err != nil {
+		return err
+	}
+	minimum := HeaderSize
+	switch h.PacketType {
+	case PacketTypeRequest, PacketTypeResponse:
+		minimum = RequestSize
+		if h.PacketFlags.IsSet(PacketFlagObjectUUID) {
+			minimum += ObjectUUIDSize
+		}
+	case PacketTypeFault:
+		minimum += 16
+	case PacketTypeBind, PacketTypeAlterContext:
+		minimum += 12
+	case PacketTypeBindAck, PacketTypeAlterContextResponse:
+		minimum += 10
+	case PacketTypeBindNak:
+		minimum += 2
+	case PacketTypeAuth3:
+		minimum += 4
+	case PacketTypeShutdown, PacketTypeCancel, PacketTypeOrphaned:
+	default:
+		return fmt.Errorf("%w: unknown packet type", ErrMalformedFragment)
+	}

+	}

+	}

+package dcerpc

+import (
+	"context"

+// boundedIO keeps buffer ownership on the calling goroutine. Cancellation closes
+// the connection and is joined before return, so no detached I/O can reuse data.
+func (t *transport) boundedIO(ctx context.Context, operation func() error) error {
+	if err := t.HasErr(); err != nil {
+		return err
+	}
+	if err := context.Cause(ctx); err != nil {
+		return err
+	}
+	if t.settings.Timeout > 0 {
+		var cancel context.CancelFunc
+		ctx, cancel = context.WithTimeout(ctx, t.settings.Timeout)
+		defer cancel()
+	}
+	joined := make(chan struct{})
+	stop := context.AfterFunc(ctx, func() {
+		defer close(joined)
+		t.retire(context.Cause(ctx))
+	})
+	err := operation()
+	if !stop() {
+		<-joined
+	}
+	err = errors.Join(err, context.Cause(ctx))
+	if err != nil {
+		return t.retire(err)
+	}
+	return nil
+}
+
+func (t *transport) retire(err error) error {
+	t.WithErr(err)
+	t.retireOnce.Do(func() { _ = t.cc.Close() })
+	return err
+}
+
+// readExact rejects a stalled/nonconforming Reader instead of spinning forever.
+func readExact(reader io.Reader, buffer []byte) error {
+	for read := 0; read < len(buffer); {
+		n, err := reader.Read(buffer[read:])
+		if n < 0 || n > len(buffer)-read {
+			return ErrMalformedFragment
+		}
+		read += n
+		if read == len(buffer) {
+			return nil
+		}
+		if err != nil {
+			if errors.Is(err, io.EOF) && read > 0 {
+				return errors.Join(io.ErrUnexpectedEOF, err)
+			}
+			return err
+		}
+		if n == 0 {
+			return io.ErrNoProgress
+		}
+	}
+	return nil
+}

+func (s *protectedStream) retire(err error) error {
+	s.transport.retire(err)

+	s.receive.Destroy()

+// exchange is reachable only from the future authenticated adapter, not the SDK
+// dispatcher. A synthetic pre-established record context can exercise this seam.
+func (s *protectedStream) exchange(ctx context.Context, opnum uint16, stub []byte) ([]byte, error) {
+	select {
+	case s.gate <- struct{}{}:
+	case <-ctx.Done():
+		return nil, context.Cause(ctx)
+	}
+	defer func() { <-s.gate }()
+	if err := context.Cause(ctx); err != nil {
+		return nil, err
+	}
+	if err := s.transport.HasErr(); err != nil {
+		return nil, err
+	}
+	if len(stub) > maxProtectedStub || s.callID == math.MaxUint32 {
+		return nil, s.retire(ErrPacketTooLong)
+	}
+	if s.transport.settings.Timeout > 0 {
+		var cancel context.CancelFunc
+		ctx, cancel = context.WithTimeout(ctx, s.transport.settings.Timeout)
+		defer cancel()
+	}
+	s.callID++
+	remaining := stub
+	first := true
+	for fragments := 0; first || len(remaining) > 0; fragments++ {
+		if fragments >= maxProtectedFragments {
+			return nil, s.retire(ErrPacketTooLong)
+		}
+		signatureSize := s.send.Size(ctx, true)
+		capacity := s.transport.settings.MaxXmitFrag - RequestSize - SecurityTrailerSize - signatureSize - 15
+		if capacity < 1 || signatureSize < 16 || signatureSize > 512 {
+			return nil, s.retire(ErrBufferTooSmall)
+		}
+		count := min(len(remaining), capacity)
+		last := count == len(remaining)
+		if err := s.writeRequest(ctx, opnum, remaining[:count], first, last); err != nil {
+			return nil, s.retire(err)
+		}
+		remaining = remaining[count:]
+		first = false
+	}
+	var result []byte
+	first = true
+	for fragments := 0; ; fragments++ {
+		if fragments >= maxProtectedFragments {
+			return result, s.retire(ErrPacketTooLong)
+		}
+		frame := make([]byte, s.transport.settings.MaxRecvFrag)
+		hdr, err := s.transport.ReadBuffer(ctx, frame)
+		if err != nil {
+			return result, s.retire(err)
+		}
+		frame = frame[:hdr.FragLength]
+		part, last, err := s.readResponse(ctx, hdr, frame, first)
+		if err != nil {
+			return result, s.retire(err)
+		}
+		if len(part) > maxProtectedStub-len(result) {
+			return result, s.retire(ErrPacketTooLong)
+		}
+		if len(part) == 0 && !last {
+			return result, s.retire(ErrMalformedFragment)
+		}
+		result = append(result, part...)
+		clear(part)
+		if last {
+			return result, nil
+		}
+		first = false
+	}

+func (s *protectedStream) writeRequest(ctx context.Context, opnum uint16, stub []byte, first, last bool) error {
+	if s.sendSequence == math.MaxUint64 {
+		return ErrNoSecurityContext
+	}
+	pad := (-len(stub)) & 15
+	length := RequestSize + len(stub) + pad + SecurityTrailerSize + s.send.Size(ctx, true)

+		return ErrPacketTooLong
+	}
+	frame := make([]byte, length)
+	defer clear(frame)
+	frame[0], frame[2], frame[4] = 5, byte(PacketTypeRequest), 0x10
+	if first {
+		frame[3] |= byte(PacketFlagFirstFrag)
+	}
+	if last {
+		frame[3] |= byte(PacketFlagLastFrag)
+	}
+	binary.LittleEndian.PutUint16(frame[8:], uint16(length))
+	binary.LittleEndian.PutUint16(frame[10:], uint16(s.send.Size(ctx, true)))
+	binary.LittleEndian.PutUint32(frame[12:], s.callID)
+	binary.LittleEndian.PutUint32(frame[16:], uint32(len(stub)))
+	binary.LittleEndian.PutUint16(frame[20:], s.presentationID)
+	binary.LittleEndian.PutUint16(frame[22:], opnum)

+	end := RequestSize + len(stub) + pad

+	binary.LittleEndian.PutUint32(trailer[4:], s.securityID)

+		return err
+	}
+	if len(token) != s.send.Size(ctx, true) {
+		return ErrNoSecurityContext
+	}
+	copy(frame[end+SecurityTrailerSize:], token)
+	var header Header
+	if err := header.ReadFrom(ctx, s.transport.Codec(frame[:HeaderSize], s.transport.settings.DataRepresentation)); err != nil {
+		return err
+	}
+	if err := s.transport.WriteBuffer(ctx, header, frame); err != nil {
+		return err
+	}
+	s.sendSequence++
+	return nil

+	} else if hdr.PacketType != PacketTypeResponse {

+		return nil, false, ErrNoSecurityContext

+	}

+func qualificationResponse(ctx context.Context, cipher rpcRecordCipher, call uint32, sequence uint64, flags byte, body []byte, fault bool) ([]byte, error) {
+	headerSize := 24
+	ptype := PacketTypeResponse
+	if fault {
+		headerSize = 32
+		ptype = PacketTypeFault
+	}
+	pad := (-len(body)) & 15
+	end := headerSize + len(body) + pad
+	frame := make([]byte, end+8+cipher.Size(ctx, true))
+	frame[0], frame[2], frame[3], frame[4] = 5, byte(ptype), flags, 0x10
+	binary.LittleEndian.PutUint16(frame[8:], uint16(len(frame)))
+	binary.LittleEndian.PutUint16(frame[10:], uint16(cipher.Size(ctx, true)))
+	binary.LittleEndian.PutUint32(frame[12:], call)
+	binary.LittleEndian.PutUint32(frame[16:], 0xffffffff) // untrusted hint is never an allocation request
+	binary.LittleEndian.PutUint16(frame[20:], 3)
+	if fault {
+		binary.LittleEndian.PutUint32(frame[24:], 5)
+	}
+	copy(frame[headerSize:], body)
+	trailer := frame[end : end+8]
+	trailer[0], trailer[1], trailer[2] = 16, 6, byte(pad)
+	binary.LittleEndian.PutUint32(trailer[4:], 9)
+	sealed := frame[headerSize:end]
+	token, err := cipher.WrapEx(ctx, sequence, [][]byte{frame[:headerSize], sealed, trailer}, [][]byte{sealed})
+	if err != nil {
+		return nil, err
+	}
+	copy(frame[end+8:], token)
+	return frame, nil
+}

+func qualificationReadRequest(ctx context.Context, tr *transport, cipher rpcRecordCipher) ([]byte, uint32, error) {
+	var result []byte
+	var call uint32
+	for sequence := uint64(0); ; sequence++ {
+		frame := make([]byte, tr.settings.MaxRecvFrag)
+		hdr, err := tr.ReadBuffer(ctx, frame)
+		if err != nil {
+			return nil, 0, err
+		}
+		if hdr.PacketType != PacketTypeRequest || hdr.PacketFlags.IsSet(PacketFlagFirstFrag) != (sequence == 0) {
+			return nil, 0, ErrMalformedFragment
+		}
+		if sequence == 0 {
+			call = hdr.CallID
+		} else if call != hdr.CallID {
+			return nil, 0, ErrMalformedFragment
+		}
+		frame = frame[:hdr.FragLength]
+		end := len(frame) - int(hdr.AuthLength) - 8
+		trailer := frame[end : end+8]
+		body := frame[24:end]
+		ok, err := cipher.UnwrapEx(ctx, sequence, [][]byte{frame[:24], body, trailer}, [][]byte{body}, frame[end+8:])
+		if err != nil || !ok {
+			return nil, 0, errors.Join(ErrNoSecurityContext, err)
+		}
+		if binary.LittleEndian.Uint16(frame[22:]) != 19 {
+			return nil, 0, ErrMalformedFragment
+		}
+		result = append(result, body[:len(body)-int(trailer[2])]...)
+		if hdr.PacketFlags.IsSet(PacketFlagLastFrag) {
+			return result, call, nil
+		}
+	}

+func (c *cursor) take(size, alignment int) []byte {
+	if c.err != nil {
+		return nil
+	}
+	if err := context.Cause(c.ctx); err != nil {
+		c.err = err
+		return nil

+		c.err = ErrMalformedNDR
+		return nil
+	}
+	c.offset += pad
+	result := c.data[c.offset : c.offset+size]
+	c.offset += size
+	return result
+}

+// DecodeReplicationInfo accepts a complete authenticated response stub. RPC faults
+// and truncated protected streams must be handled separately, before this decoder.
+func DecodeReplicationInfo(ctx context.Context, requested uint32, stub []byte) (reply Reply, err error) {
+	if requested != 0 && requested != 3 && requested != 4 && requested != 5 {
+		return reply, ErrMalformedNDR
+	}
+	if len(stub) > maxStubBytes {
+		return reply, ErrNDRLimit
+	}
+	c := &cursor{ctx: ctx, data: stub}
+	reply.Kind = c.u32()
+	discriminant := c.u32()
+	reply.Present = c.u32() != 0
+	if c.err != nil {
+		return reply, c.err
+	}
+	if discriminant != reply.Kind || reply.Kind != 0 && reply.Kind != 3 && reply.Kind != 4 && reply.Kind != 5 {
+		return reply, ErrMalformedNDR
+	}
+	var semantic error
+	if reply.Present {
+		conformance := c.u32()
+		minimum := 36
+		switch reply.Kind {
+		case 0:
+			c.take(0, 8)
+			reply.Count = c.u32()
+			c.u32()
+			minimum = 128
+		case 3, 4:
+			reply.Count = c.u32()
+			c.u32()
+		case 5:
+			reply.Started = c.filetime()
+			reply.Count = c.u32()
+			minimum = 68
+		}
+		if c.err != nil {
+			return reply, c.err
+		}
+		if conformance > maxRecords || reply.Count > maxRecords {
+			return reply, ErrNDRLimit
+		}
+		if conformance != reply.Count {
+			semantic = ErrMalformedNDR
+		}
+		if int(conformance) > (len(stub)-c.offset)/minimum {
+			return reply, ErrMalformedNDR
+		}
+		switch reply.Kind {

+		case 3, 4:

+		}
+		if c.err != nil {
+			return reply, errors.Join(semantic, c.err)
+		}
+	}
+	status := c.u32()
+	if c.err != nil {
+		return reply, errors.Join(semantic, c.err)
+	}

+	}

+	}

+	}

+func writeHash(writer io.Writer, parts ...any) error {
+	write := func(value []byte) error {
+		n, err := writer.Write(value)
+		if err == nil && n != len(value) {

+		}
+		return err

+				return err

+				}

+package crypto
+
+// maxRecordBytes bounds plaintext, signed headers and encrypted data before allocation.
+const maxRecordBytes = 16 << 20
+
+func sameBuffer(a, b []byte) bool {
+	return len(a) == len(b) && (len(a) == 0 || &a[0] == &b[0])
+}
+
+// validateRecordBuffers requires each sealed buffer to appear exactly once in the
+// signing order. This prevents unauthenticated ciphertext and ambiguous aliases.
+func validateRecordBuffers(sign, seal [][]byte) error {
+	total := 0
+	for _, b := range sign {
+		if len(b) > maxRecordBytes-total {
+			return ErrMessageTooLarge
+		}
+		total += len(b)
+	}
+	for i, b := range seal {
+		if len(b) == 0 {
+			continue
+		}
+		matches := 0
+		for _, signed := range sign {
+			if sameBuffer(b, signed) {
+				matches++
+			}
+		}
+		if matches != 1 {
+			return ErrInvalidToken
+		}
+		for _, previous := range seal[:i] {
+			if sameBuffer(previous, b) {
+				return ErrInvalidToken
+			}
+		}
+	}
+	return nil
+}
