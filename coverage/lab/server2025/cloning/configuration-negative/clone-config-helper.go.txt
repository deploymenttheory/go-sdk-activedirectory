package main

import (
	"context"
	"encoding/json"
	"errors"
	"flag"
	"fmt"
	"io"
	"os"
	"path/filepath"
	"strings"
	"syscall"
	"time"

	"github.com/deploymenttheory/go-sdk-activedirectory/dccloning"
)

type nativeError struct {
	Operation string `json:"operation"`
	Code      uint32 `json:"code"`
}

type errorDetails struct {
	Present              bool         `json:"present"`
	Message              string       `json:"message"`
	Permission           bool         `json:"permission"`
	Incomplete           bool         `json:"incomplete"`
	InvalidConfiguration bool         `json:"invalid_configuration"`
	InvalidPath          bool         `json:"invalid_path"`
	Prerequisite         bool         `json:"prerequisite"`
	NotDomainController  bool         `json:"not_domain_controller"`
	OutputExists         bool         `json:"output_exists"`
	Unsupported          bool         `json:"unsupported"`
	Deadline             bool         `json:"deadline"`
	NativeError          *nativeError `json:"native_error"`
	ErrnoCode            *uint32      `json:"errno_code"`
}

func describeError(err error) errorDetails {
	details := errorDetails{
		Present:              err != nil,
		Permission:           errors.Is(err, os.ErrPermission),
		Incomplete:           errors.Is(err, dccloning.ErrIncomplete),
		InvalidConfiguration: errors.Is(err, dccloning.ErrInvalidConfiguration),
		InvalidPath:          errors.Is(err, dccloning.ErrInvalidPath),
		Prerequisite:         errors.Is(err, dccloning.ErrClonePrerequisite),
		NotDomainController:  errors.Is(err, dccloning.ErrNotDomainController),
		OutputExists:         errors.Is(err, dccloning.ErrOutputExists),
		Unsupported:          errors.Is(err, dccloning.ErrUnsupported),
		Deadline:             errors.Is(err, context.DeadlineExceeded),
	}
	if err == nil {
		return details
	}
	details.Message = err.Error()
	var native *dccloning.NativeError
	if errors.As(err, &native) {
		details.NativeError = &nativeError{Operation: native.Operation, Code: native.Code}
	}
	var errno syscall.Errno
	if errors.As(err, &errno) {
		code := uint32(errno)
		details.ErrnoCode = &code
	}
	return details
}

func readOptions(path string) (options dccloning.ConfigOptions, resultErr error) {
	file, err := os.Open(path)
	if err != nil {
		return options, err
	}
	defer func() { resultErr = errors.Join(resultErr, file.Close()) }()
	info, err := file.Stat()
	if err != nil {
		return options, err
	}
	if !info.Mode().IsRegular() || info.Size() > 65536 {
		return options, errors.New("options must be a regular file of at most 64 KiB")
	}
	decoder := json.NewDecoder(io.LimitReader(file, 65537))
	decoder.DisallowUnknownFields()
	start, err := decoder.Token()
	if err != nil {
		return options, err
	}
	if start != json.Delim('{') {
		return options, errors.New("options must be an object")
	}
	seen := map[string]bool{}
	for decoder.More() {
		token, err := decoder.Token()
		if err != nil {
			return options, err
		}
		key, ok := token.(string)
		if !ok {
			return options, errors.New("options key must be a string")
		}
		key = strings.ToLower(key)
		if seen[key] {
			return options, fmt.Errorf("duplicate options field %q", key)
		}
		seen[key] = true
		var target any
		switch key {
		case "configuration":
			target = &options.Configuration
		case "directory":
			target = &options.Directory
		case "offline":
			target = &options.Offline
		case "timeout":
			target = &options.Timeout
		default:
			return options, fmt.Errorf("unknown options field %q", key)
		}
		var raw json.RawMessage
		if err := decoder.Decode(&raw); err != nil {
			return options, err
		}
		if string(raw) == "null" {
			return options, fmt.Errorf("null options field %q", key)
		}
		fieldDecoder := json.NewDecoder(strings.NewReader(string(raw)))
		fieldDecoder.DisallowUnknownFields()
		if err := fieldDecoder.Decode(target); err != nil {
			return options, err
		}
	}
	if end, err := decoder.Token(); err != nil || end != json.Delim('}') {
		return options, errors.New("incomplete options object")
	}
	var trailing any
	if err := decoder.Decode(&trailing); err != io.EOF {
		return options, fmt.Errorf("expected one options object: %v", err)
	}
	if !seen["offline"] || !seen["directory"] || strings.TrimSpace(options.Directory) == "" || !filepath.IsAbs(options.Directory) {
		return options, errors.New("fixture requires explicit Offline and absolute Directory")
	}
	return options, nil
}

func main() {
	path := flag.String("options", "", "owned ConfigOptions JSON fixture")
	flag.Parse()
	if *path == "" || flag.NArg() != 0 {
		fmt.Fprintln(os.Stderr, "exactly one -options path is required")
		os.Exit(2)
	}
	options, err := readOptions(*path)
	if err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(2)
	}
	ctx, cancel := context.WithTimeout(context.Background(), time.Minute)
	defer cancel()
	data, err := dccloning.GenerateConfig(ctx, options)
	output := struct {
		ProtocolVersion int                  `json:"protocol_version"`
		Operation       string               `json:"operation"`
		Data            dccloning.Configured `json:"data"`
		ErrorDetails    errorDetails         `json:"error_details"`
	}{1, "generate_config", data, describeError(err)}
	encoder := json.NewEncoder(os.Stdout)
	encoder.SetIndent("", "  ")
	if encodeErr := encoder.Encode(output); encodeErr != nil {
		fmt.Fprintln(os.Stderr, encodeErr)
		os.Exit(1)
	}
	if err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(1)
	}
}
